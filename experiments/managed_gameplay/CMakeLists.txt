cmake_minimum_required(VERSION 3.24)
project(PoimaManagedLab LANGUAGES CXX)
set(POIMA_DOTNET_HOST_HEADERS "" CACHE PATH "SDK directory containing hostfxr.h and coreclr_delegates.h")
if(NOT EXISTS "${POIMA_DOTNET_HOST_HEADERS}/hostfxr.h")
    message(FATAL_ERROR "Set POIMA_DOTNET_HOST_HEADERS to the .NET SDK's native host headers.")
endif()
add_executable(poima-managed-lab launcher.cpp)
add_library(poima_managed_kernel SHARED kernel.cpp)
foreach(target poima-managed-lab poima_managed_kernel)
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(MINGW)
        target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++)
    endif()
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
    endif()
endforeach()
target_include_directories(poima-managed-lab PRIVATE "${POIMA_DOTNET_HOST_HEADERS}")
target_link_libraries(poima-managed-lab PRIVATE ${CMAKE_DL_LIBS})
if(MINGW)
    target_link_options(poima-managed-lab PRIVATE -municode)
endif()
