cmake_minimum_required(VERSION 3.24)
project(Poima VERSION 0.0.36 LANGUAGES C CXX)

option(POIMA_ENABLE_VULKAN_PROBE "Build the optional Vulkan device inspection" ON)
option(POIMA_BUILD_TESTS "Build and register CLI contract tests (Python is test-only)" ON)
option(POIMA_BUILD_RENDER_SMOKE "Build the SDL/NVRHI Vulkan rendering experiment" OFF)
option(POIMA_BUILD_EDITOR "Build the optional native SDL/Vulkan authoring editor" OFF)
option(POIMA_BUILD_DESKTOP_BRIDGE "Build the optional native desktop frontend bridge" OFF)
option(POIMA_BUILD_MODULE_LAB "Build the isolated native reload experiment" ON)
option(POIMA_ENABLE_SIMULATION "Build the Jolt/EnTT fixed-step runtime" OFF)
option(POIMA_ENABLE_AUDIO "Build native Steam Audio observation and capture" OFF)
option(POIMA_ENABLE_MANAGED_GAMEPLAY "Host CoreCLR gameplay in the native runtime" OFF)
option(POIMA_ENABLE_NATIVE_GAMEPLAY "Load process-lived Native AOT gameplay libraries" ${POIMA_ENABLE_SIMULATION})
set(POIMA_DOTNET_HOST_HEADERS "" CACHE PATH "Directory containing hostfxr.h and coreclr_delegates.h")
if(POIMA_BUILD_RENDER_SMOKE AND NOT POIMA_ENABLE_VULKAN_PROBE)
    message(FATAL_ERROR "Rendering requires POIMA_ENABLE_VULKAN_PROBE=ON.")
endif()
if(POIMA_BUILD_EDITOR AND NOT POIMA_BUILD_RENDER_SMOKE)
    message(FATAL_ERROR "The editor requires POIMA_BUILD_RENDER_SMOKE=ON.")
endif()

if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The bootstrap currently targets Windows and Linux only.")
endif()

string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" POIMA_TARGET_ARCH)
if(POIMA_TARGET_ARCH MATCHES "^(amd64|x86_64|x64)$")
    set(POIMA_TARGET_ARCH x86_64)
endif()
configure_file(cmake/build_info.hpp.in generated/poima/build_info.hpp @ONLY)
add_library(poima_core STATIC src/core.cpp src/host.cpp src/world.cpp src/scene.cpp src/player.cpp src/input_profile.cpp src/input_profile_store.cpp src/hash.cpp src/gltf.cpp src/model_codec.cpp src/texture.cpp src/tangents.cpp src/lighting.cpp src/shadows.cpp src/visibility.cpp src/gameplay.cpp src/audio_clip.cpp src/audio.cpp src/audio_events.cpp src/audio_stream.cpp src/animation.cpp)
target_sources(poima_core PRIVATE src/runtime_animation.cpp src/runtime_geometry.cpp src/editor_gizmo.cpp)
add_library(poima_mikktspace STATIC third_party/mikktspace/mikktspace.c)
set_target_properties(poima_mikktspace PROPERTIES C_STANDARD 99 C_EXTENSIONS OFF)
target_link_libraries(poima_core PRIVATE poima_mikktspace)
target_compile_features(poima_core PUBLIC cxx_std_20)
set_target_properties(poima_core PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(poima_core PUBLIC include PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(poima_core SYSTEM PRIVATE third_party)
target_sources(poima_core PRIVATE src/local_session.cpp src/shared_session.cpp src/project.cpp src/game_launch.cpp src/native_gameplay_artifact.cpp)
if(WIN32)
    target_link_libraries(poima_core PRIVATE advapi32)
endif()
target_compile_definitions(poima_core PRIVATE POIMA_AUDIO=$<BOOL:${POIMA_ENABLE_AUDIO}>)
if(POIMA_ENABLE_AUDIO)
    include(cmake/audio.cmake)
endif()
target_compile_definitions(poima_core PRIVATE POIMA_SIMULATION=$<BOOL:${POIMA_ENABLE_SIMULATION}>)
target_compile_definitions(poima_core PRIVATE POIMA_MANAGED_GAMEPLAY=$<BOOL:${POIMA_ENABLE_MANAGED_GAMEPLAY}>)
target_compile_definitions(poima_core PRIVATE POIMA_NATIVE_GAMEPLAY=$<BOOL:${POIMA_ENABLE_NATIVE_GAMEPLAY}>)
if(POIMA_ENABLE_NATIVE_GAMEPLAY AND NOT POIMA_ENABLE_SIMULATION)
    message(FATAL_ERROR "Native gameplay requires simulation.")
endif()
if(POIMA_ENABLE_NATIVE_GAMEPLAY OR POIMA_ENABLE_MANAGED_GAMEPLAY)
    target_link_libraries(poima_core PRIVATE ${CMAKE_DL_LIBS})
endif()
if(POIMA_ENABLE_MANAGED_GAMEPLAY)
    if(NOT POIMA_ENABLE_SIMULATION OR NOT EXISTS "${POIMA_DOTNET_HOST_HEADERS}/hostfxr.h")
        message(FATAL_ERROR "Managed gameplay requires simulation and POIMA_DOTNET_HOST_HEADERS from a .NET SDK.")
    endif()
    target_include_directories(poima_core SYSTEM PRIVATE "${POIMA_DOTNET_HOST_HEADERS}")
    target_link_libraries(poima_core PRIVATE ${CMAKE_DL_LIBS})
endif()
if(POIMA_ENABLE_SIMULATION)
    include(cmake/simulation.cmake)
else()
    target_sources(poima_core PRIVATE src/runtime_disabled.cpp)
endif()

if(POIMA_ENABLE_VULKAN_PROBE)
    include(FetchContent)
    set(VULKAN_HEADERS_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
    set(VULKAN_HEADERS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(vulkan_headers
        URL https://codeload.github.com/KhronosGroup/Vulkan-Headers/tar.gz/015e25c3c91b70eb1a754d36fb14c4ba6ad9b0b9
        URL_HASH SHA256=666700236101d288dea8bb080c6f8039467cb1d29ea1e1aae6f93a819f4f9029
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(vulkan_headers)
    target_sources(poima_core PRIVATE src/vulkan_probe.cpp)
    target_link_libraries(poima_core PRIVATE Vulkan::Headers ${CMAKE_DL_LIBS})
else()
    target_sources(poima_core PRIVATE src/vulkan_probe_disabled.cpp)
endif()
target_compile_definitions(poima_core PRIVATE POIMA_VULKAN_PROBE=$<BOOL:${POIMA_ENABLE_VULKAN_PROBE}>)
target_compile_definitions(poima_core PRIVATE POIMA_RENDER_SMOKE=$<BOOL:${POIMA_BUILD_RENDER_SMOKE}>)
target_compile_definitions(poima_core PRIVATE POIMA_EDITOR=$<BOOL:${POIMA_BUILD_EDITOR}>)

if(POIMA_BUILD_RENDER_SMOKE)
    include(cmake/render_smoke.cmake)
else()
    target_sources(poima_core PRIVATE src/render_smoke_disabled.cpp src/gamepad_disabled.cpp src/hosted_viewport_disabled.cpp)
endif()

if(POIMA_BUILD_EDITOR)
    include(cmake/editor.cmake)
else()
    target_sources(poima_core PRIVATE src/editor_disabled.cpp src/editor_viewport_disabled.cpp)
endif()

if(POIMA_BUILD_DESKTOP_BRIDGE)
    include(cmake/desktop_bridge.cmake)
endif()

add_executable(poima src/main.cpp)
if(MINGW)
    target_link_options(poima PRIVATE -municode)
endif()
target_link_libraries(poima PRIVATE poima_core)
set_target_properties(poima PROPERTIES CXX_EXTENSIONS OFF)

if(POIMA_BUILD_MODULE_LAB)
    add_executable(poima-module-lab experiments/native_module/host.cpp)
    target_compile_features(poima-module-lab PRIVATE cxx_std_20)
    target_include_directories(poima-module-lab PRIVATE include)
    target_link_libraries(poima-module-lab PRIVATE ${CMAKE_DL_LIBS})
    if(WIN32)
        target_link_libraries(poima-module-lab PRIVATE psapi)
    endif()
    if(MSVC)
        target_compile_options(poima-module-lab PRIVATE /W4 /permissive- /utf-8)
    else()
        target_compile_options(poima-module-lab PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endif()

foreach(target poima_core poima)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    endif()
endforeach()
